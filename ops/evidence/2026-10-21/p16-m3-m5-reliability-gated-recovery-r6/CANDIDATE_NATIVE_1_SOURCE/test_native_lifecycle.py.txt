"""One registered synthetic native-host capability exercise, not a scientific fit."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

from . import native_process as native

R = Path(__file__).resolve().parents[1]
P = R / 'paper16_development_r6'
E = R / 'ops/evidence/2026-10-21/p16-m3-m5-reliability-gated-recovery-r6'
W = P / 'work/native_lifecycle_1'
MODULE = 'paper16_development_r6.test_native_lifecycle'


def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def J(path):
    return json.loads(path.read_bytes())


def once(path, value):
    with path.open('x', encoding='utf8', newline='\n') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write('\n')
        stream.flush()
        os.fsync(stream.fileno())


def guard():
    reg = J(E/'NATIVE_LIFECYCLE_REGISTRATION_1.json')
    assert reg['phase_claim_sha256'] == sha(P/'PHASE_A_CLAIM.json')
    claim = J(P/'PHASE_A_CLAIM.json')
    assert claim['start_monotonic'] <= time.monotonic() < claim['deadline_monotonic']
    assert datetime.datetime.now(datetime.timezone.utc) < datetime.datetime.fromisoformat(claim['deadline_UTC'])
    for row in reg['source_files']:
        assert sha(R/row['path']) == row['sha256']
    return reg


def command(role, *args):
    return [sys.executable, '-B', '-m', MODULE, role, *args]


def wait_file(path, seconds=15):
    end = time.monotonic() + seconds
    while not path.exists():
        guard()
        if time.monotonic() >= end:
            raise TimeoutError('Registered synthetic receipt missing: '+path.name)
        time.sleep(.02)
    return J(path)


def launcher():
    guard()
    once(W/'LAUNCHER.json', {'pid': os.getpid(), 'parent_pid': os.getppid(), **native.current_job()})
    with (W/'BROKER_STDOUT.log').open('xb') as log:
        try:
            broker = native.launch_detached(command('broker'), R, log, breakaway=True)
        except OSError as exc:
            once(W/'BREAKAWAY_DENIAL.json', {'winerror': exc.winerror, 'errno': exc.errno,
                 'message': str(exc), 'child_created': False, 'retry_performed': False,
                 'observed_host_capability_failure': True})
            return 19
        code = broker.wait(timeout=15)
    once(W/'LAUNCHER_COMPLETE.json', {'broker_pid': broker.pid, 'broker_actual_exit_code': code,
                                     'launcher_pid': os.getpid(), 'no_retry': True})
    return code


def broker():
    guard()
    context = native.current_job()
    assert context['in_job'] is False, 'Broker remains in inherited Job'
    once(W/'BROKER.json', {'pid': os.getpid(), 'parent_pid': os.getppid(), **context})
    with (W/'SUPERVISOR_STDOUT.log').open('xb') as log:
        supervisor = native.launch_detached(command('supervisor'), R, log, breakaway=False)
    once(W/'BROKER_COMPLETE.json', {'pid': os.getpid(), 'supervisor_pid': supervisor.pid,
                                   'supervisor_handle_not_retained': True})
    return 0


def worker(mode):
    guard()
    native.require_job(os.environ['P16_R6_SYNTHETIC_JOB'])
    if mode == 'normal':
        once(W/'NORMAL_DONE.json', {'pid': os.getpid(), 'synthetic_complete': True,
                                    'job_name': os.environ['P16_R6_SYNTHETIC_JOB'],
                                    'native_job': native.current_job()})
        return 0
    if mode == 'nonzero':
        return 17
    if mode == 'empty_zero':
        return 0
    if mode == 'grandchild':
        once(W/'GRANDCHILD.json', {'pid': os.getpid()})
        time.sleep(30)
        return 0
    if mode == 'tree':
        child = subprocess.Popen(command('worker', 'grandchild'), cwd=R,
                                 stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        once(W/'TREE_WORKER.json', {'pid': os.getpid(), 'grandchild_pid': child.pid})
        time.sleep(30)
        return 0
    if mode == 'loss_tree':
        child = subprocess.Popen(command('worker', 'grandchild_loss'), cwd=R,
                                 stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        once(W/'LOSS_WORKER.json', {'pid': os.getpid(), 'grandchild_pid': child.pid})
        time.sleep(30)
        return 0
    if mode == 'grandchild_loss':
        once(W/'LOSS_GRANDCHILD.json', {'pid': os.getpid()})
        time.sleep(30)
        return 0
    raise ValueError('Unregistered fixture role')


def supervisor():
    guard()
    context = native.current_job()
    assert context['in_job'] is False, 'Detached supervisor remains in inherited Job'
    broker = wait_file(W/'BROKER_COMPLETE.json')
    launcher = wait_file(W/'LAUNCHER.json')
    native.wait_not_alive(broker['pid'], 10)
    native.wait_not_alive(launcher['pid'], 10)
    once(W/'DETACHED_LIVENESS.json', {'pid': os.getpid(), 'parent_pid': os.getppid(),
         'broker_pid': broker['pid'], 'launcher_pid': launcher['pid'],
         'broker_dead': True, 'launcher_dead': True, **context})
    wait_file(W/'CONTINUE_SYNTHETIC.json')
    outcomes = {}
    for mode in ['normal', 'nonzero', 'empty_zero', 'tree']:
        outcomes[mode] = native.supervise(command('worker', mode), R, W/(mode+'.log'),
                            'Local\\P16_R6_SYNTHETIC_'+str(os.getpid())+'_'+mode, 2.0 if mode=='tree' else 10.0)
    tree = wait_file(W/'TREE_WORKER.json')
    native.wait_not_alive(tree['pid'], 5)
    native.wait_not_alive(tree['grandchild_pid'], 5)
    normal = outcomes['normal']['returncode'] == 0 and (W/'NORMAL_DONE.json').exists()
    nonzero = outcomes['nonzero']['returncode'] == 17
    empty = outcomes['empty_zero']['returncode'] == 0 and not (W/'EMPTY_DONE.json').exists()
    killed = outcomes['tree']['killed_at_deadline'] is True
    once(W/'CONTAINMENT_CASES.json', {'outcomes': outcomes, 'normal_pass': normal,
          'nonzero_failclosed': nonzero, 'zero_without_receipt_failclosed': empty,
          'deadline_descendant_kill_pass': killed, 'all_cases_pass': normal and nonzero and empty and killed})
    wait_file(W/'INJECT_SUPERVISOR_LOSS.json')
    # The driver deliberately kills this owned supervisor. Its Job handle then
    # closes, so its synthetic worker/grandchild must also die.
    native.supervise(command('worker', 'loss_tree'), R, W/'loss_tree.log',
                     'Local\\P16_R6_SYNTHETIC_'+str(os.getpid())+'_loss', 15.0)
    raise RuntimeError('Injected supervisor loss did not occur')


def exercise():
    reg = guard()
    assert not W.exists()
    W.mkdir(parents=True)
    once(W/'EXERCISE_CLAIM.json', {'registration_sha256': sha(E/'NATIVE_LIFECYCLE_REGISTRATION_1.json'),
          'phase_claim_sha256': reg['phase_claim_sha256'], 'driver_pid': os.getpid(), 'one_exercise': True,
          'synthetic_only': True, 'retries': 0, 'start_UTC': datetime.datetime.now(datetime.timezone.utc).isoformat()})
    result = {'schema': 'R6_NATIVE_LIFECYCLE_RESULT_V1', 'mission': reg['mission'],
              'registration_sha256': sha(E/'NATIVE_LIFECYCLE_REGISTRATION_1.json'),
              'driver_context': native.current_job(), 'synthetic_only': True, 'real_A_calls': 0,
              'E_Final_access': False, 'OS_settings_modified': False, 'scientific_stage_started': False}
    with (W/'LAUNCHER_STDOUT.log').open('xb') as log:
        proc = subprocess.Popen(command('launcher'), cwd=R, stdout=log, stderr=subprocess.STDOUT)
        try:
            code = proc.wait(timeout=30)
        except BaseException:
            if proc.poll() is None:
                proc.kill()
            proc.wait()
            raise
    result['launcher_actual_exit_code'] = code
    if (W/'BREAKAWAY_DENIAL.json').exists():
        result.update(host_capability_pass=False, mandatory_host_gate='FAIL',
                      failure=J(W/'BREAKAWAY_DENIAL.json'), remaining_cases_executed=False,
                      scientific_status='STOP_HOST_RELIABILITY_REQUIRED_GATE')
    else:
        assert code == 0
        live = wait_file(W/'DETACHED_LIVENESS.json')
        assert live['launcher_dead'] and live['broker_dead'] and not live['in_job']
        assert native.process_alive(live['pid'])
        once(W/'CONTINUE_SYNTHETIC.json', {'driver_observed_live_after_broker_and_launcher_exit': True})
        outcomes = wait_file(W/'CONTAINMENT_CASES.json', 30)
        assert outcomes['all_cases_pass']
        once(W/'INJECT_SUPERVISOR_LOSS.json', {'target_supervisor_pid': live['pid']})
        loss = wait_file(W/'LOSS_WORKER.json')
        wait_file(W/'LOSS_GRANDCHILD.json')
        killer = subprocess.run(['taskkill.exe', '/PID', str(live['pid']), '/F'],
                                stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        assert killer.returncode == 0
        native.wait_not_alive(live['pid'], 10)
        native.wait_not_alive(loss['pid'], 10)
        native.wait_not_alive(loss['grandchild_pid'], 10)
        once(W/'INJECTED_LOSS_STOP.json', {'injected_synthetic_loss': True, 'no_PASS': True, 'no_retry': True})
        try:
            once(W/'EXERCISE_CLAIM.json', {})
        except FileExistsError:
            duplicate_denied = True
        else:
            duplicate_denied = False
        assert duplicate_denied
        result.update(host_capability_pass=True, mandatory_host_gate='PASS_TESTED_CAPABILITIES_ONLY',
                      detached_liveness=live, containment=outcomes, injected_loss_failclosed=True,
                      duplicate_claim_denied=True, scientific_stage_started=False)
    result['files'] = [{'path': f.relative_to(R).as_posix(), 'sha256': sha(f), 'bytes': f.stat().st_size}
                       for f in sorted(W.iterdir()) if f.is_file()]
    result['completed_UTC'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    once(E/'NATIVE_LIFECYCLE_RESULT_1.json', result)
    print(json.dumps({'host_capability_pass': result['host_capability_pass'],
                      'mandatory_host_gate': result['mandatory_host_gate'],
                      'launcher_actual_exit_code': code,
                      'result_sha256': sha(E/'NATIVE_LIFECYCLE_RESULT_1.json')}, indent=2))
    return 0 if result['host_capability_pass'] else 19


if __name__ == '__main__':
    role = sys.argv[1]
    action = {'exercise': exercise, 'launcher': launcher, 'broker': broker, 'supervisor': supervisor}
    raise SystemExit(worker(sys.argv[2]) if role == 'worker' else action[role]())
